using UnityEngine;using UnityEditor;using System.Linq;
internal class CommandScript:IRunCommand{public void Execute(ExecutionResult result){foreach(var f in Object.FindObjectsByType<MeshFilter>(FindObjectsInactive.Include,FindObjectsSortMode.None).Where(x=>x.name=="Surface_176"||x.name=="Context_124")){var m=f.sharedMesh;var v=m.vertices;var n=m.normals;var ids=Enumerable.Range(0,m.subMeshCount).Select(i=>m.GetTriangles(i)).ToArray();m.vertices=v;m.normals=n;for(int i=0;i<ids.Length;i++)m.SetTriangles(ids[i],i);m.RecalculateBounds();m.UploadMeshData(false);EditorUtility.SetDirty(m);}AssetDatabase.SaveAssets();result.Log("Reuploaded identical serialized vertex/index arrays; no geometry/source coordinates changed");}}
