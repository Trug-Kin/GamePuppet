static bool ClosestAttribute(Vector3 p,Vector3[] v,int[] ts,out Vector3Int ids,out Vector3 weights,out float distance){ids=default;weights=default;float best=.002f*.002f;bool found=false;for(int i=0;i<ts.Length;i+=3){Vector3 a=v[ts[i]]-p,b=v[ts[i+1]]-p,c=v[ts[i+2]]-p;var lo=Vector3.Min(a,Vector3.Min(b,c));var hi=Vector3.Max(a,Vector3.Max(b,c));if(lo.x>.002f||lo.y>.002f||lo.z>.002f||hi.x<-.002f||hi.y<-.002f||hi.z<-.002f)continue;Vector3 candidate=default,bary=default;float dbest=float.PositiveInfinity;var ab=b-a;var ac=c-a;var n=Vector3.Cross(ab,ac);if(n.sqrMagnitude>1e-20f){var q=n*(Vector3.Dot(a,n)/n.sqrMagnitude);double aa=Vector3.Dot(ab,ab),bb=Vector3.Dot(ab,ac),cc=Vector3.Dot(ac,ac),dd=Vector3.Dot(q-a,ab),ee=Vector3.Dot(q-a,ac);double den=aa*cc-bb*bb;if(System.Math.Abs(den)>1e-20){double u=(cc*dd-bb*ee)/den,w=(aa*ee-bb*dd)/den;if(u>=0&&w>=0&&u+w<=1){candidate=q;bary=new Vector3((float)(1-u-w),(float)u,(float)w);dbest=q.sqrMagnitude;}}}for(int e=0;e<3;e++){var start=e==0?a:e==1?b:c;var end=e==0?b:e==1?c:a;var line=end-start;float t=line.sqrMagnitude>1e-20f?Mathf.Clamp01(-Vector3.Dot(start,line)/line.sqrMagnitude):0;var q=start+line*t;if(q.sqrMagnitude<dbest){dbest=q.sqrMagnitude;candidate=q;bary=e==0?new Vector3(1-t,t,0):e==1?new Vector3(0,1-t,t):new Vector3(t,0,1-t);}}if(dbest<best){best=dbest;ids=new Vector3Int(ts[i],ts[i+1],ts[i+2]);weights=bary;found=true;}}distance=Mathf.Sqrt(best);return found;}
