using UnityEngine;using UnityEditor;using System.IO;using System.Text;using System.Linq;
internal class CommandScript:IRunCommand{public void Execute(ExecutionResult r){var b=new StringBuilder();foreach(var t in Object.FindObjectsByType<TextMesh>(FindObjectsInactive.Include,FindObjectsSortMode.None))if(t.text.ToLowerInvariant().Contains("hotel")||t.text.ToLowerInvariant().Contains("bình")||t.text.ToLowerInvariant().Contains("xưởng"))b.AppendLine(P(t.transform)+" text="+t.text+" world="+t.transform.position);foreach(var t in Object.FindObjectsByType<Transform>(FindObjectsInactive.Include,FindObjectsSortMode.None))if(t.name.Contains("Hotel")||t.name.Contains("Gate")||t.name.Contains("NhuY"))if(t.GetComponent<Renderer>())b.AppendLine(P(t)+" "+t.GetComponent<Renderer>().bounds);File.WriteAllText("Test/KetQua/investigation_r107/landmarks.txt",b.ToString());r.Log(b.ToString().Substring(0,Mathf.Min(2500,b.Length)));}string P(Transform t)=>t.parent?P(t.parent)+"/"+t.name:t.name;}
