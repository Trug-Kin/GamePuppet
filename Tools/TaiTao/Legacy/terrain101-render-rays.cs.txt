using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using System.IO;using System.Linq;
internal class CommandScript:IRunCommand{static string P(Transform t)=>t.parent?P(t.parent)+"/"+t.name:t.name;public void Execute(ExecutionResult result){var b=new System.Text.StringBuilder();var fs=EditorSceneManager.GetActiveScene().GetRootGameObjects().SelectMany(g=>g.GetComponentsInChildren<MeshFilter>(true)).Where(f=>f.sharedMesh&&f.gameObject.activeInHierarchy&&f.GetComponent<MeshRenderer>()&&f.GetComponent<MeshRenderer>().enabled).ToArray();foreach(var pos in new[]{new Vector3(648,10,414),new Vector3(650,10,414),new Vector3(646,10,417.5f),new Vector3(648,10,417.9f),new Vector3(648,10,418.1f)}){foreach(var f in fs){var bounds=f.GetComponent<MeshRenderer>().bounds;if(pos.x<bounds.min.x||pos.x>bounds.max.x||pos.z<bounds.min.z||pos.z>bounds.max.z)continue;var g=new GameObject("ReadOnlyRenderRay");g.hideFlags=HideFlags.HideAndDontSave;g.transform.SetParent(f.transform.parent,false);g.transform.localPosition=f.transform.localPosition;g.transform.localRotation=f.transform.localRotation;g.transform.localScale=f.transform.localScale;try{var c=g.AddComponent<MeshCollider>();c.sharedMesh=f.sharedMesh;Physics.SyncTransforms();RaycastHit h;if(c.Raycast(new Ray(pos,Vector3.down),out h,30))b.AppendLine("render "+pos.x+","+pos.z+" y="+h.point.y+" "+P(f.transform));}finally{Object.DestroyImmediate(g);}}}File.WriteAllText("Test/KetQua/journey_v48/terrain_r101/render-layer-rays.txt",b.ToString());result.Log(b.ToString());}}
