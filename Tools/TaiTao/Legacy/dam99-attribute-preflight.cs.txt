using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using UnityEngine.Rendering;using System.IO;using System.Linq;using System.Collections.Generic;
internal class CommandScript:IRunCommand{
static string P(Transform t)=>t.parent?P(t.parent)+"/"+t.name:t.name;
public void Execute(ExecutionResult result){var sc=EditorSceneManager.GetActiveScene();if(!sc.path.EndsWith("r98.unity")||EditorApplication.isPlaying)throw new System.Exception("R98 Edit required");var all=sc.GetRootGameObjects().SelectMany(g=>g.GetComponentsInChildren<Transform>(true)).GroupBy(t=>P(t)).ToDictionary(g=>g.Key,g=>g.First());var log=new System.Text.StringBuilder();foreach(var l in File.ReadAllLines("Test/KetQua/journey_v48/dam_r99/integration.tsv")){var e=l.Split('\t');if(e[3]=="1"||e[4]=="1")continue;var t=all[e[2]];var m=t.GetComponent<MeshFilter>().sharedMesh;var r=t.GetComponent<MeshRenderer>();log.AppendLine(e[1]+"\t"+e[2]+"\t"+m.vertexCount+"\tcolor="+m.HasVertexAttribute(VertexAttribute.Color)+"\tuv1="+m.HasVertexAttribute(VertexAttribute.TexCoord1)+"\t"+string.Join(";",r.sharedMaterials.Select(x=>x?x.name+":"+x.shader.name:"NULL")));}File.WriteAllText("Test/KetQua/journey_v48/dam_r99/attribute-preflight.txt",log.ToString());result.Log(sc.path+" clean="+!sc.isDirty+"; written attributes preflight");}}
