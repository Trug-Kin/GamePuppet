using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using System;using System.IO;using System.Linq;using System.Text;using ConRoi.Investigation.V36.R104;
internal class CommandScript:IRunCommand{
const string A="Assets/Art/InvestigationV36/R104/";
static Transform Bone(Transform root,params string[] names)=>root.GetComponentsInChildren<Transform>(true).FirstOrDefault(t=>names.Contains(t.name));
static void Layer(Transform root,int layer){foreach(var t in root.GetComponentsInChildren<Transform>(true))t.gameObject.layer=layer;}
static Material Mat(Material source){if(!source)return null;string path=A+"Materials/"+source.name+".mat";var m=AssetDatabase.LoadAssetAtPath<Material>(path);if(!m){m=new Material(Shader.Find("Universal Render Pipeline/Lit"));m.name=source.name;m.color=source.HasProperty("_BaseColor")?source.GetColor("_BaseColor"):source.color;m.SetFloat("_Smoothness",.13f);AssetDatabase.CreateAsset(m,path);}return m;}
static GameObject Art(string file,Transform parent){var asset=AssetDatabase.LoadAssetAtPath<GameObject>(A+file+".fbx");if(!asset)throw new Exception("Absent "+file);var g=(GameObject)PrefabUtility.InstantiatePrefab(asset);g.name="Art_R104_"+file;g.transform.SetParent(parent,false);foreach(var a in g.GetComponentsInChildren<Animation>())a.enabled=false;foreach(var a in g.GetComponentsInChildren<Animator>())a.enabled=false;foreach(var rr in g.GetComponentsInChildren<Renderer>())rr.sharedMaterials=rr.sharedMaterials.Select(Mat).ToArray();return g;}
static void Replace(InvestigationActor actor,string file,AnimationClip[] poses,AnimationClip repair){
 if(actor.nativeRoot)throw new Exception("Art already bound "+actor.name);
 foreach(Transform child in actor.transform.Cast<Transform>().ToArray())if(child.GetComponentInChildren<SkinnedMeshRenderer>(true)||child.GetComponentsInChildren<Transform>(true).Any(t=>t.name=="head"))Object.DestroyImmediate(child.gameObject);
 var g=Art(file,actor.transform);actor.nativeRoot=g.transform;actor.nativeIdle=AssetDatabase.LoadAllAssetsAtPath(A+file+".fbx").OfType<AnimationClip>().First(c=>!c.name.StartsWith("__preview__"));
 actor.head=Bone(g.transform,"Head","head");actor.leftUpper=Bone(g.transform,"UpperArm.L","upperL");actor.rightUpper=Bone(g.transform,"UpperArm.R","upperR");actor.leftFore=Bone(g.transform,"LowerArm.L","foreL");actor.rightFore=Bone(g.transform,"LowerArm.R","foreR");if(actor.puppet){actor.poseClips=poses;actor.repairClip=repair;}
 actor.nativeIdle.SampleAnimation(g,0);Layer(g.transform,actor.gameObject.layer);
 if(actor.name=="Khai_Investigation"||actor.name=="R03_Investigation")g.transform.localPosition=Vector3.down*.08f/actor.transform.localScale.y;
}
static void String(string name,Transform parent,Vector3 a,Vector3 b,Material material){var go=new GameObject(name);go.transform.SetParent(parent,false);var line=go.AddComponent<LineRenderer>();line.useWorldSpace=false;line.positionCount=2;line.SetPosition(0,a);line.SetPosition(1,b);line.startWidth=line.endWidth=.008f;line.sharedMaterial=material;}
public void Execute(ExecutionResult r){var sc=EditorSceneManager.GetActiveScene();if(EditorApplication.isPlaying||sc.isDirty||!sc.path.EndsWith("r104.unity"))throw new Exception("Clean Edit r104 required");var s=Object.FindFirstObjectByType<InvestigationSession>();
foreach(string file in new[]{"Khai","Ha","Puppet","PuppetPoseLibrary104"}){AssetDatabase.ImportAsset(A+file+".fbx",ImportAssetOptions.ForceSynchronousImport);var imp=(ModelImporter)AssetImporter.GetAtPath(A+file+".fbx");imp.animationType=ModelImporterAnimationType.Legacy;imp.importAnimation=true;imp.isReadable=true;imp.SaveAndReimport();}
AssetDatabase.ImportAsset(A+"R12Support.fbx",ImportAssetOptions.ForceSynchronousImport);
var clips=AssetDatabase.LoadAllAssetsAtPath(A+"PuppetPoseLibrary104.fbx").OfType<AnimationClip>().Where(c=>!c.name.StartsWith("__preview__")).ToArray();var poses=InvestigationContent.Alphabet.Select(ch=>clips.Single(c=>c.name.Contains("Pose_"+ch+"_R104"))).ToArray();var repair=clips.Single(c=>c.name.Contains("RepairBothHands_R104"));
var actors=s.GetComponentsInChildren<InvestigationActor>(true).Where(a=>a.transform.parent.name!="Reconstruction_stage").ToArray();foreach(var actor in actors)Replace(actor,actor.puppet?"Puppet":actor.name.StartsWith("Ha")?"Ha":"Khai",poses,repair);
// Frame and suspension support R12 in the clip; retain the neutral Tư event performer.
var clip=s.clipGate.transform.parent;var support=Art("R12Support",clip);support.transform.localPosition=new Vector3(.35f,3,1.1f);Layer(support.transform,30);
var ropeMat=AssetDatabase.LoadAssetAtPath<Material>(A+"Materials/Paper ivory.mat");foreach(var pair in new[]{new[]{"head","head"},new[]{"upperL","left"},new[]{"upperR","right"}}){var end=Bone(s.clipRepair,pair[0]);if(end)String("R12_suspension_"+pair[1],clip,new Vector3(.35f+(pair[1]=="left"?.35f:pair[1]=="right"?-.35f:0),5.1f,1.45f),clip.InverseTransformPoint(end.position),ropeMat);}
foreach(var b in s.repairEventActor.GetComponents<RepairHandBinding>()){b.session=s;b.repairAttachment=Bone(s.clipRepair,"handR");if(!b.repairAttachment)throw new Exception("R12 right hand");}
var mainR12=actors.Single(a=>a.name=="R12_Investigation"&&a.transform.parent.name=="Workshop_GDD36");var mainSupport=Art("R12Support",mainR12.transform.parent);mainSupport.transform.localPosition=mainR12.transform.localPosition;
// Store render textures at the same aspect as their camera presentation and enlarge pixel resolution.
foreach(var texture in new[]{s.clipTexture,s.stageTexture}){texture.width=1600;texture.height=1000;EditorUtility.SetDirty(texture);}
s.clipCamera.transform.localPosition=new Vector3(2.65f,4.75f,-1.10f);s.clipCamera.transform.LookAt(clip.TransformPoint(new Vector3(.82f,4.10f,1.52f)));s.clipCamera.fieldOfView=30;
s.data=new InvestigationSave();s.diagnostics=false;s.UpdateClip();s.clipCamera.enabled=false;
var text=new StringBuilder("R104 art bound: Khai, Ha, R03, R12, cipher puppet, clip R12. Stage/event Tư remain declared neutral prototype performers.\n");foreach(var actor in actors)text.AppendLine(actor.name+" -> "+actor.nativeRoot.name+" idle="+actor.nativeIdle.name+" pose refs="+actor.poseClips.Length);text.AppendLine("15 retargeted shared pose IDs; repair pose="+repair.name);EditorSceneManager.MarkSceneDirty(sc);EditorSceneManager.SaveScene(sc);AssetDatabase.SaveAssets();File.WriteAllText("Test/KetQua/investigation_r104/art-integration.txt",text.ToString());r.Log(text.ToString());}}
