using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using System;using System.IO;using System.Linq;using System.Collections.Generic;
internal class CommandScript:IRunCommand{static string P(Transform t)=>t.parent?P(t.parent)+"/"+t.name:t.name;
[Serializable]class Row{public string path,asset,colliderAsset,prefab,mesh,colliderType;public string[] materials=new string[0];public bool active,render=false,collider=true,readable;public int layer,vertices;public float[] min,max,matrix;}
public void Execute(ExecutionResult result){if(EditorApplication.isPlaying)throw new Exception("Edit only");string d="Test/KetQua/journey_v48/ground_r93/";var all=EditorSceneManager.GetActiveScene().GetRootGameObjects().SelectMany(x=>x.GetComponentsInChildren<Collider>(true)).ToArray();File.WriteAllText(d+"primitive-colliders.txt",string.Join("\n",all.Where(c=>!(c is MeshCollider)&&c.bounds.max.x>450&&c.bounds.min.x<870&&c.bounds.max.z>40&&c.bounds.min.z<610).Select(c=>P(c.transform)+"|"+c.GetType().Name+"|active="+c.gameObject.activeInHierarchy+"|trigger="+c.isTrigger+"|layer="+c.gameObject.layer+"|"+c.bounds)));
var pick=all.OfType<MeshCollider>().Where(c=>c.enabled&&c.gameObject.activeInHierarchy&&c.sharedMesh&&c.sharedMesh.isReadable&&c.bounds.max.x>450&&c.bounds.min.x<870&&c.bounds.max.z>40&&c.bounds.min.z<610&&c.bounds.min.y<25&&c.bounds.size.x*c.bounds.size.z>15).ToArray();var rows=new List<Row>();using(var w=new BinaryWriter(File.Create(d+"scene-colliders.meshbin"))){w.Write(0x523537);w.Write(pick.Length);foreach(var c in pick){var m=c.sharedMesh;string path=P(c.transform)+"#COLLIDER";var b=c.bounds;var mat=c.transform.localToWorldMatrix;var a=new float[16];for(int j=0;j<16;j++)a[j]=mat[j];rows.Add(new Row{path=path,asset=AssetDatabase.GetAssetPath(m),colliderAsset=AssetDatabase.GetAssetPath(m),colliderType="MeshCollider",prefab=PrefabUtility.GetPrefabAssetPathOfNearestInstanceRoot(c.gameObject),mesh=m.name,active=true,readable=true,layer=c.gameObject.layer,vertices=m.vertexCount,min=new[]{b.min.x,b.min.y,b.min.z},max=new[]{b.max.x,b.max.y,b.max.z},matrix=a});w.Write(path);w.Write(AssetDatabase.GetAssetPath(m));for(int j=0;j<16;j++)w.Write(mat[j]);var v=m.vertices;w.Write(v.Length);w.Write(m.subMeshCount);w.Write(0);w.Write(0);w.Write(0);foreach(var p in v){w.Write(p.x);w.Write(p.y);w.Write(p.z);}for(int j=0;j<m.subMeshCount;j++){var t=m.GetTriangles(j);w.Write(t.Length);foreach(int x in t)w.Write(x);}}}File.WriteAllText(d+"collider-inventory.json","{\"rows\":["+string.Join(",",rows.Select(r=>JsonUtility.ToJson(r)))+"]}");result.Log("actual collider mesh dump="+pick.Length);}}
