using UnityEngine;using UnityEditor;using UnityEngine.InputSystem;using System.IO;using System.Linq;
internal class CommandScript:IRunCommand{static string P(Transform t)=>t.parent?P(t.parent)+"/"+t.name:t.name;public void Execute(ExecutionResult result){var s=Object.FindFirstObjectByType<DieuPhoiHanhTrinh>();var b=new System.Text.StringBuilder();b.AppendLine("region="+s.data.region+" playerActive="+s.player.gameObject.activeInHierarchy+" enabled="+s.player.enabled+" paused="+s.Paused+" timescale="+Time.timeScale+" state="+JsonUtility.ToJson(s.data).Substring(0,400));b.AppendLine("actions "+string.Join(";",InputSystem.ListEnabledActions().Select(a=>(a.actionMap!=null?a.actionMap.name:"standalone")+"/"+a.name)));foreach(var r in Object.FindObjectsByType<MeshRenderer>(FindObjectsInactive.Include,FindObjectsSortMode.None)){if(!r.enabled||!r.gameObject.activeInHierarchy)continue;if(!r.name.ToLower().Contains("flower")&&!r.sharedMaterials.Any(m=>m&&(m.name.Contains("Petal")||m.name.Contains("Flower"))))continue;if(r.bounds.max.x<630||r.bounds.min.x>695||r.bounds.max.z<480||r.bounds.min.z>535)continue;b.AppendLine(P(r.transform)+" min="+r.bounds.min+" max="+r.bounds.max+" materials="+string.Join(";",r.sharedMaterials.Select(m=>m?m.name:"NULL")));}File.WriteAllText("Test/KetQua/journey_v48/dam_r99/play-inspect.txt",b.ToString());result.Log(b.ToString().Substring(0,System.Math.Min(1200,b.Length)));}}
