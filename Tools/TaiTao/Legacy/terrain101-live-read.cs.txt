using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using System.IO;using System.Linq;
internal class CommandScript:IRunCommand{static string P(Transform t)=>t.parent?P(t.parent)+"/"+t.name:t.name;public void Execute(ExecutionResult result){var log=new System.Text.StringBuilder();var sc=EditorSceneManager.GetActiveScene();foreach(var t in sc.GetRootGameObjects().SelectMany(g=>g.GetComponentsInChildren<Transform>(true))){var f=t.GetComponent<MeshFilter>();var c=t.GetComponent<MeshCollider>();var r=t.GetComponent<MeshRenderer>();if(!(r&&r.bounds.min.x<647&&r.bounds.max.x>640&&r.bounds.min.z<418&&r.bounds.max.z>410)&&!(c&&c.bounds.min.x<647&&c.bounds.max.x>640&&c.bounds.min.z<418&&c.bounds.max.z>410))continue;log.AppendLine(P(t)+" active="+t.gameObject.activeInHierarchy+" layer="+t.gameObject.layer+" render="+(r?r.enabled.ToString():"none")+" mesh="+(f&&f.sharedMesh?AssetDatabase.GetAssetPath(f.sharedMesh):"none")+" collider="+(c?c.enabled+" "+AssetDatabase.GetAssetPath(c.sharedMesh):"none")+" scale="+t.lossyScale);if(t.name=="Surface_142"){var m=f.sharedMesh;log.AppendLine("readable="+m.isReadable+" vertex="+m.vertexCount+" colliderSame="+(c&&c.sharedMesh==m));var vs=m.vertices;var ids=m.triangles;int near=0,down=0;for(int i=0;i<ids.Length;i+=3){var a=t.TransformPoint(vs[ids[i]]);var b=t.TransformPoint(vs[ids[i+1]]);var d=t.TransformPoint(vs[ids[i+2]]);var mid=(a+b+d)/3;if(mid.x<640||mid.x>650||mid.z<409||mid.z>418)continue;near++;if(Vector3.Cross(b-a,d-a).y<0)down++;}log.AppendLine("local faces="+near+" downward="+down);}}File.WriteAllText("Test/KetQua/journey_v48/terrain_r101/live-bindings.txt",log.ToString());result.Log(log.ToString());}}
