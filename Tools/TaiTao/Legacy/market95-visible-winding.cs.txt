using UnityEngine;using UnityEditor;using System.IO;using System.Linq;using System.Text;
internal class CommandScript:IRunCommand{public void Execute(ExecutionResult result){var b=new StringBuilder();foreach(var f in Object.FindObjectsByType<MeshFilter>(FindObjectsInactive.Include,FindObjectsSortMode.None).Where(x=>x.name=="Surface_176"||x.name=="Context_124")){var m=f.sharedMesh;var v=m.vertices;var t=m.triangles;foreach(var p in new[]{new Vector2(702,420),new Vector2(696,413),new Vector2(700,415)})for(int i=0;i<t.Length;i+=3){var a=f.transform.TransformPoint(v[t[i]]);var bb=f.transform.TransformPoint(v[t[i+1]]);var c=f.transform.TransformPoint(v[t[i+2]]);float den=(bb.z-c.z)*(a.x-c.x)+(c.x-bb.x)*(a.z-c.z);if(Mathf.Abs(den)<1e-8)continue;float u=((bb.z-c.z)*(p.x-c.x)+(c.x-bb.x)*(p.y-c.z))/den,w=((c.z-a.z)*(p.x-c.x)+(a.x-c.x)*(p.y-c.z))/den;if(u<0||w<0||u+w>1)continue;b.AppendLine(f.name+" xz="+p+" triangle="+i/3+" worldCross="+Vector3.Cross(bb-a,c-a).normalized+" normal="+f.transform.TransformDirection(m.normals[t[i]])+" y="+(u*a.y+w*bb.y+(1-u-w)*c.y));}}File.WriteAllText("Test/KetQua/journey_v48/market_ground_r95/visible-winding.txt",b.ToString());result.Log(b.ToString());}}
