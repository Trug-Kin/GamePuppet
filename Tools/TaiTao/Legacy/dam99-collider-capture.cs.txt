using UnityEngine;using Unity.Collections;using UnityEditor;using UnityEditor.SceneManagement;using UnityEngine.Rendering;using System.IO;using System.Linq;using System.Collections.Generic;
internal class CommandScript:IRunCommand{static GameObject Find(string path){var a=EditorSceneManager.GetActiveScene().GetRootGameObjects();var names=path.Split('/');var t=a.SingleOrDefault(x=>x.name==names[0])?.transform;for(int i=1;t&&i<names.Length;i++)t=t.Find(names[i]);return t?t.gameObject:null;}
static UnityEngine.Mesh Readable(UnityEngine.Mesh src){if(src.isReadable)return src;using(var data=UnityEditor.MeshUtility.AcquireReadOnlyMeshData(src)){var d=data[0];var m=new UnityEngine.Mesh();m.indexFormat=src.indexFormat;using(var a=new NativeArray<Vector3>(d.vertexCount,Allocator.Temp)){d.GetVertices(a);m.vertices=a.ToArray();}if(src.HasVertexAttribute(VertexAttribute.Normal))using(var a=new NativeArray<Vector3>(d.vertexCount,Allocator.Temp)){d.GetNormals(a);m.normals=a.ToArray();}for(int ch=0;ch<8;ch++){var attr=(VertexAttribute)((int)VertexAttribute.TexCoord0+ch);if(!src.HasVertexAttribute(attr))continue;int dim=src.GetVertexAttributeDimension(attr);if(dim==2){using(var a=new NativeArray<Vector2>(d.vertexCount,Allocator.Temp)){d.GetUVs(ch,a);m.SetUVs(ch,a.ToArray());}}else if(dim==3){using(var a=new NativeArray<Vector3>(d.vertexCount,Allocator.Temp)){d.GetUVs(ch,a);m.SetUVs(ch,a.ToArray());}}else using(var a=new NativeArray<Vector4>(d.vertexCount,Allocator.Temp)){d.GetUVs(ch,a);m.SetUVs(ch,a.ToArray());}}if(src.HasVertexAttribute(VertexAttribute.Color))using(var a=new NativeArray<Color>(d.vertexCount,Allocator.Temp)){d.GetColors(a);m.colors=a.ToArray();}if(src.HasVertexAttribute(VertexAttribute.Tangent))using(var a=new NativeArray<Vector4>(d.vertexCount,Allocator.Temp)){d.GetTangents(a);m.tangents=a.ToArray();}m.subMeshCount=d.subMeshCount;for(int j=0;j<d.subMeshCount;j++){var s=d.GetSubMesh(j);var ids=new int[s.indexCount];if(src.indexFormat==IndexFormat.UInt16){var raw=d.GetIndexData<ushort>();for(int i=0;i<ids.Length;i++)ids[i]=raw[s.indexStart+i]+s.baseVertex;}else{var raw=d.GetIndexData<int>();for(int i=0;i<ids.Length;i++)ids[i]=raw[s.indexStart+i]+s.baseVertex;}m.SetTriangles(ids,j);}m.bounds=src.bounds;return m;}}
public void Execute(ExecutionResult result){if(EditorApplication.isPlaying||EditorSceneManager.GetActiveScene().isDirty||!EditorSceneManager.GetActiveScene().path.EndsWith("r98.unity"))throw new System.Exception("Clean edit R98 required");string d="Test/KetQua/journey_v48/dam_r99/collider-sources/";Directory.CreateDirectory(d);var b=new System.Text.StringBuilder();int index=0;foreach(var c in Object.FindObjectsByType<MeshCollider>(FindObjectsInactive.Include,FindObjectsSortMode.None)){if(!c.sharedMesh||!c.enabled||c.isTrigger||!c.gameObject.activeInHierarchy||c.gameObject.layer!=3||c.transform.root.name=="R98_OngDao_Concrete_Dam")continue;var z=c.bounds;if(z.max.x<610||z.min.x>710||z.max.z<465||z.min.z>558)continue;var m=Readable(c.sharedMesh);string p=c.name;var t=c.transform.parent;while(t){p=t.name+"/"+p;t=t.parent;}string key="COL"+index.ToString("D3");index++;using(var w=new BinaryWriter(File.Create(d+key+".bin"))){w.Write(m.vertexCount);w.Write(m.triangles.Length);foreach(var v in m.vertices){var world=c.transform.TransformPoint(v);w.Write(world.x);w.Write(world.y);w.Write(world.z);}foreach(var i in m.triangles)w.Write(i);}var f=c.GetComponent<MeshFilter>();b.AppendLine(key+"\t"+p+"\t"+AssetDatabase.GetAssetPath(c.sharedMesh)+"\t"+(f&&f.sharedMesh==c.sharedMesh));}File.WriteAllText(d+"index.tsv",b.ToString());result.Log("Captured "+index+" actual old colliders for scoped Blender comparison; no scene changes");}}
