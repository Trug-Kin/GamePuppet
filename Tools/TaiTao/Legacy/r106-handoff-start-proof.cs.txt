var settings=InputSystem.settings;if(AssetDatabase.Contains(settings))throw new Exception("Runtime InputSettings required");string previous=JsonUtility.ToJson(settings);settings.editorInputBehaviorInPlayMode=InputSettings.EditorInputBehaviorInPlayMode.AllDeviceInputAlwaysGoesToGameView;settings.backgroundBehavior=InputSettings.BackgroundBehavior.IgnoreFocus;var kb=InputSystem.AddDevice<Keyboard>("PT001_StartCheck_Keyboard");kb.MakeCurrent();int step=0;double next=EditorApplication.timeSinceStartup+1;EditorApplication.CallbackFunction f=null;
