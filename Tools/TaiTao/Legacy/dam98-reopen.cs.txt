using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using System.IO;using System.Linq;using System.Text;
internal class CommandScript:IRunCommand{public void Execute(ExecutionResult result){
if(EditorApplication.isPlaying)throw new System.Exception("Edit required");EditorSceneManager.NewScene(NewSceneSetup.EmptyScene,NewSceneMode.Single);var sc=EditorSceneManager.OpenScene("Assets/Scenes/JourneyPresentation_v48_r98.unity");var all=sc.GetRootGameObjects().SelectMany(x=>x.GetComponentsInChildren<Transform>(true)).ToArray();var b=new StringBuilder();int bad=0;
foreach(var line in File.ReadAllLines("Test/KetQua/journey_v48/dam_r98/integration.tsv")){var e=line.Split('\t');string path=e[1]==""?"R98_OngDao_Concrete_Dam/"+e[0]:e[1];var names=path.Split('/');var t=sc.GetRootGameObjects().Single(x=>x.name==names[0]).transform;for(int i=1;i<names.Length;i++)t=t.Find(names[i]);var f=t.GetComponent<MeshFilter>();var c=t.GetComponent<MeshCollider>();var r=t.GetComponent<MeshRenderer>();string asset=AssetDatabase.GetAssetPath(f.sharedMesh);bool ok=asset.Contains("DamR98/")&&(!c||!c.enabled||c.sharedMesh==f.sharedMesh)&&r&&r.sharedMaterials.All(m=>m);if(!ok)bad++;b.AppendLine(path+" mesh="+asset+" collider="+(c?AssetDatabase.GetAssetPath(c.sharedMesh):"NONE")+" enabled="+(c&&c.enabled)+" layer="+t.gameObject.layer+" materials="+string.Join(";",r.sharedMaterials.Select(m=>m?AssetDatabase.GetAssetPath(m):"MISSING"))+" ok="+ok);}
Physics.SyncTransforms();foreach(float s in new[]{5f,15f,25f,35f,45f,52f}){var p=new Vector3(637.7344f+.76346655f*s-.64584737f*5,15,528.2603f-.64584737f*s-.76346655f*5);b.AppendLine("APRON s="+s+" d=5 "+string.Join(";",Physics.RaycastAll(p,Vector3.down,25,~0,QueryTriggerInteraction.Ignore).Where(h=>h.collider is MeshCollider).Select(h=>h.collider.name+"@"+h.point.y.ToString("F6")+":"+AssetDatabase.GetAssetPath(((MeshCollider)h.collider).sharedMesh))));}
var root=GameObject.Find("R98_OngDao_Concrete_Dam");b.AppendLine("scene="+sc.path+" dirty="+sc.isDirty+" bad="+bad+" missing="+all.Sum(t=>GameObjectUtility.GetMonoBehavioursWithMissingScriptCount(t.gameObject))+" roots98="+all.Count(t=>t.name=="R98_OngDao_Concrete_Dam")+" children="+root.transform.childCount+" rootScale="+root.transform.localScale+" prefab="+PrefabUtility.GetPrefabAssetPathOfNearestInstanceRoot(root)+" buildScenes="+string.Join(";",EditorBuildSettings.scenes.Select(s=>s.path)));File.WriteAllText("Test/KetQua/journey_v48/dam_r98/final-reopened-inspector.txt",b.ToString());result.Log("Reopened r98, inspector bad="+bad+"; scoped collider rays recorded");}}
