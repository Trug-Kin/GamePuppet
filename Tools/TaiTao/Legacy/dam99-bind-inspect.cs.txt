using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using System.IO;using System.Linq;using System.Text;
internal class CommandScript:IRunCommand{
static string PathOf(Transform t){return t.parent?PathOf(t.parent)+"/"+t.name:t.name;}static GameObject Find(string path){return EditorSceneManager.GetActiveScene().GetRootGameObjects().SelectMany(g=>g.GetComponentsInChildren<Transform>(true)).FirstOrDefault(t=>PathOf(t)==path)?.gameObject;}
public void Execute(ExecutionResult result){var b=new StringBuilder();foreach(var line in File.ReadAllLines("Test/KetQua/journey_v48/dam_r99/current-paths.tsv")){var e=line.Split('\t');var g=Find(e[1]);if(!g){b.AppendLine(line+"\tMISSING");continue;}var f=g.GetComponent<MeshFilter>();var r=g.GetComponent<MeshRenderer>();b.AppendLine(line+"\t"+g.activeInHierarchy+"\t"+(r&&r.enabled)+"\t"+(f?AssetDatabase.GetAssetPath(f.sharedMesh):"")+"\t"+string.Join(";",g.GetComponents<Collider>().Select(c=>c.GetType().Name+":"+c.enabled+":"+c.isTrigger+":"+(c is MeshCollider m?AssetDatabase.GetAssetPath(m.sharedMesh):"")))+"\t"+(r?string.Join(";",r.sharedMaterials.Select(m=>m?AssetDatabase.GetAssetPath(m):"MISSING")):"")+"\t"+g.transform.position.ToString("F5"));}File.WriteAllText("Test/KetQua/journey_v48/dam_r99/unity-binding-audit.tsv",b.ToString());result.Log("Current bindings inspected without mutation");}}
