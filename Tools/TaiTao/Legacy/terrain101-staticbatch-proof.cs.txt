using UnityEngine;using UnityEngine.Rendering;using Unity.Collections;using UnityEditor;using System.IO;using System.Linq;
internal class CommandScript:IRunCommand{
static string Key(Vector3 p)=>Mathf.RoundToInt(p.x*1000)+","+Mathf.RoundToInt(p.y*1000)+","+Mathf.RoundToInt(p.z*1000);
static string[] Keys(Vector3[] v,int[] indices)=>Enumerable.Range(0,indices.Length/3).Select(i=>string.Join("|",new[]{Key(v[indices[i*3]]),Key(v[indices[i*3+1]]),Key(v[indices[i*3+2]])}.OrderBy(s=>s))).OrderBy(s=>s).ToArray();
static bool BatchedSame(Transform t,UnityEngine.Mesh expected,out string detail){var r=t.GetComponent<MeshRenderer>();var m=t.GetComponent<MeshFilter>().sharedMesh;if(!r.isPartOfStaticBatch){detail="not static batched";return m==expected;}var so=new SerializedObject(r);var info=so.FindProperty("m_StaticBatchInfo");int first=r.subMeshStartIndex,count=info.FindPropertyRelative("subMeshCount").intValue;Vector3[] vs;var indices=new System.Collections.Generic.List<int>();using(var md=MeshUtility.AcquireReadOnlyMeshData(m)){var d=md[0];using(var nv=new NativeArray<Vector3>(d.vertexCount,Allocator.Temp)){d.GetVertices(nv);vs=nv.ToArray();}for(int j=first;j<first+count;j++){var sm=d.GetSubMesh(j);for(int k=0;k<sm.indexCount;k++)indices.Add((m.indexFormat==IndexFormat.UInt16?d.GetIndexData<ushort>()[sm.indexStart+k]:d.GetIndexData<int>()[sm.indexStart+k])+sm.baseVertex);}}
var ev=expected.vertices.Select(v=>t.TransformPoint(v)).ToArray();bool same=Keys(vs,indices.ToArray()).SequenceEqual(Keys(ev,expected.triangles));detail="staticBatch=true first="+first+" count="+count+" actualTriangles="+indices.Count/3+" expectedTriangles="+expected.triangles.Length/3+" worldTriangleKeysMatch1mm="+same+" transform="+t.localToWorldMatrix;return same;}
public void Execute(ExecutionResult result){var b=new System.Text.StringBuilder();foreach(var name in new[]{"R97_LOWER_WALK","R97_LOWER_GRASS"}){var t=Object.FindObjectsByType<Transform>(FindObjectsInactive.Include,FindObjectsSortMode.None).Single(t=>t.name==name);var e=AssetDatabase.LoadAssetAtPath<UnityEngine.Mesh>("Assets/Art/Environment/JourneyV48/TerrainR101/"+name+"_IntegratedR101.asset");string detail;bool same=BatchedSame(t,e,out detail);b.AppendLine(name+" "+detail);if(!same)throw new System.Exception("Actual batched shape differs "+name);}File.WriteAllText("Test/KetQua/journey_v48/terrain_r101/static-batch-proof.txt",b.ToString());result.Log(b.ToString());}}
