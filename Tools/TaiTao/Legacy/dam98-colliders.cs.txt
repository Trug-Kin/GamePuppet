using UnityEngine;using UnityEditor;using System.IO;using System.Linq;using System.Text;
internal class CommandScript:IRunCommand{public void Execute(ExecutionResult result){var b=new StringBuilder();foreach(var c in Object.FindObjectsByType<MeshCollider>(FindObjectsInactive.Include,FindObjectsSortMode.None)){if(!c.sharedMesh||!c.enabled)continue;var z=c.bounds;if(z.max.x<620||z.min.x>706||z.max.z<478||z.min.z>552)continue;string p=c.name;var t=c.transform.parent;while(t){p=t.name+"/"+p;t=t.parent;}b.AppendLine(p+" | "+z.min.ToString("F4")+".."+z.max.ToString("F4")+" | "+AssetDatabase.GetAssetPath(c.sharedMesh)+" active="+c.gameObject.activeInHierarchy);if(p.StartsWith("OngDao_V35_Photo_Batch/OD35_COLLIDER_")){var m=c.sharedMesh;b.AppendLine("vertices "+string.Join(";",m.vertices.Select(v=>c.transform.TransformPoint(v).ToString("F6"))));}}File.WriteAllText("Test/KetQua/journey_v48/dam_r98/colliders.txt",b.ToString());result.Log("Collider details saved");}}
