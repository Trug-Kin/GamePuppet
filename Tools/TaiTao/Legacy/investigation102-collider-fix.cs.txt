using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using System;using System.IO;using System.Text;using ConRoi.Investigation.V36;
internal class CommandScript:IRunCommand{public void Execute(ExecutionResult r){if(EditorApplication.isPlaying)throw new Exception("Edit only");var scene=EditorSceneManager.GetActiveScene();if(!scene.path.EndsWith("r102.unity")||scene.isDirty)throw new Exception("Clean r102 only");var session=Object.FindFirstObjectByType<InvestigationSession>();var log=new StringBuilder("Local-space collider correction (render geometry unchanged)\n");foreach(var i in session.GetComponentsInChildren<InvestigationInteractable>(true)){var c=i.GetComponent<BoxCollider>();if(!c)continue;var bounds=new Bounds();bool first=true;foreach(var renderer in i.GetComponentsInChildren<Renderer>(true)){if(renderer is LineRenderer)continue;var b=renderer.localBounds;for(int k=0;k<8;k++){var p=b.center+Vector3.Scale(b.extents,new Vector3((k&1)==0?-1:1,(k&2)==0?-1:1,(k&4)==0?-1:1));p=i.transform.InverseTransformPoint(renderer.transform.TransformPoint(p));if(first){bounds=new Bounds(p,Vector3.zero);first=false;}else bounds.Encapsulate(p);}}if(first)continue;var old=c.bounds; c.center=bounds.center;c.size=new Vector3(Mathf.Max(.06f,bounds.size.x),Mathf.Max(.04f,bounds.size.y),Mathf.Max(.06f,bounds.size.z));log.AppendLine(i.name+" before="+old+" after="+c.bounds);}EditorSceneManager.MarkSceneDirty(scene);EditorSceneManager.SaveScene(scene);File.WriteAllText("Test/KetQua/investigation_r102/collider-local-fix.txt",log.ToString());r.Log(log.ToString());}}
