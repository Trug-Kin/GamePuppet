using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using System;using System.IO;using System.Linq;using System.Text;using System.Collections.Generic;
internal class CommandScript:IRunCommand{
static string P(Transform t)=>t.parent?P(t.parent)+"/"+t.name:t.name;
[Serializable]class Row{public string path,asset,colliderAsset,prefab,mesh,colliderType;public string[] materials;public bool active,render,collider,readable;public int layer,vertices;public float[] min,max,matrix;}
public void Execute(ExecutionResult result){var sc=EditorSceneManager.GetActiveScene();if(EditorApplication.isPlaying||sc.isDirty||!sc.path.EndsWith("r93.unity"))throw new Exception("Require clean r93 Edit scene");string d="Test/KetQua/journey_v48/ground_r93/";Directory.CreateDirectory(d);var rows=new List<Row>();var pick=new List<MeshFilter>();
foreach(var f in sc.GetRootGameObjects().SelectMany(x=>x.GetComponentsInChildren<MeshFilter>(true))){if(!f.sharedMesh)continue;var m=f.sharedMesh;var rr=f.GetComponent<Renderer>();var cc=f.GetComponent<Collider>();var b=m.bounds;var pts=new[]{new Vector3(b.min.x,b.min.y,b.min.z),new Vector3(b.min.x,b.min.y,b.max.z),new Vector3(b.min.x,b.max.y,b.min.z),new Vector3(b.min.x,b.max.y,b.max.z),new Vector3(b.max.x,b.min.y,b.min.z),new Vector3(b.max.x,b.min.y,b.max.z),new Vector3(b.max.x,b.max.y,b.min.z),new Vector3(b.max.x,b.max.y,b.max.z)}.Select(p=>f.transform.TransformPoint(p)).ToArray();var wb=new Bounds(pts[0],Vector3.zero);foreach(var p in pts)wb.Encapsulate(p);if(wb.max.x<450||wb.min.x>870||wb.max.z<40||wb.min.z>610)continue;var mat=f.transform.localToWorldMatrix;var a=new float[16];for(int j=0;j<16;j++)a[j]=mat[j];var row=new Row{path=P(f.transform),asset=AssetDatabase.GetAssetPath(m),mesh=m.name,colliderAsset=cc is MeshCollider?AssetDatabase.GetAssetPath(((MeshCollider)cc).sharedMesh):"",colliderType=cc?cc.GetType().Name:"",prefab=PrefabUtility.GetPrefabAssetPathOfNearestInstanceRoot(f.gameObject),active=f.gameObject.activeInHierarchy,render=rr&&rr.enabled,collider=cc&&cc.enabled,readable=m.isReadable,layer=f.gameObject.layer,vertices=m.vertexCount,materials=rr?rr.sharedMaterials.Select(x=>x?AssetDatabase.GetAssetPath(x):"MISSING").ToArray():new string[0],min=new[]{wb.min.x,wb.min.y,wb.min.z},max=new[]{wb.max.x,wb.max.y,wb.max.z},matrix=a};rows.Add(row);
string path=row.path;bool env=path.StartsWith("R20_")||path.StartsWith("LakeGateMarket")||path.StartsWith("RouteSurfaces")||path.StartsWith("R11_")||path.StartsWith("R21_")||path.StartsWith("R22_")||path.StartsWith("OD35")||path.Contains("Garden");bool broad=wb.size.x*wb.size.z>15;bool ground=path.Contains("Surface_")||path.Contains("Context_")||path.Contains("Restored_")||path.Contains("RoadLoop")||path.Contains("Terrain")||path.Contains("Paving")||path.Contains("Footway")||path.Contains("Island")||path.Contains("Garden")||path.Contains("Bridge")||path.Contains("Planter_Sidewalk");if(env&&ground&&broad&&wb.min.y<25&&m.isReadable&&(wb.max.x<580||wb.min.z>620||wb.max.z<70))pick.Add(f);}
File.WriteAllText(d+"expanded-inventory.json","{\"scene\":\""+sc.path+"\",\"rows\":["+string.Join(",",rows.Select(x=>JsonUtility.ToJson(x)))+"]}");
using(var w=new BinaryWriter(File.Create(d+"expanded-ground.meshbin"))){w.Write(0x523537);w.Write(pick.Count);foreach(var f in pick){var m=f.sharedMesh;w.Write(P(f.transform));w.Write(AssetDatabase.GetAssetPath(m));var mat=f.transform.localToWorldMatrix;for(int j=0;j<16;j++)w.Write(mat[j]);var v=m.vertices;var n=m.normals;var uv=m.uv;var t=m.tangents;w.Write(v.Length);w.Write(m.subMeshCount);w.Write(n.Length);w.Write(uv.Length);w.Write(t.Length);foreach(var x in v){w.Write(x.x);w.Write(x.y);w.Write(x.z);}foreach(var x in n){w.Write(x.x);w.Write(x.y);w.Write(x.z);}foreach(var x in uv){w.Write(x.x);w.Write(x.y);}foreach(var x in t){w.Write(x.x);w.Write(x.y);w.Write(x.z);w.Write(x.w);}for(int j=0;j<m.subMeshCount;j++){var ids=m.GetTriangles(j);w.Write(ids.Length);foreach(int x in ids)w.Write(x);}}}
File.WriteAllText(d+"expanded-components.txt",string.Join("\n",sc.GetRootGameObjects().SelectMany(x=>x.GetComponentsInChildren<MonoBehaviour>(true)).Where(x=>x).Select(x=>P(x.transform)+"|"+x.GetType().Name).Distinct()));result.Log("scene="+sc.path+" inventory="+rows.Count+" dumped="+pick.Count+" bytes="+new FileInfo(d+"expanded-ground.meshbin").Length);}}
