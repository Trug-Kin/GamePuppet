using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using System.IO;using System.Linq;
internal class CommandScript:IRunCommand{static string P(Transform t)=>t.parent?P(t.parent)+"/"+t.name:t.name;
public void Execute(ExecutionResult result){var sc=EditorSceneManager.GetActiveScene();if(EditorApplication.isPlaying||!sc.path.EndsWith("r99.unity")||sc.isDirty)throw new System.Exception("Clean R99 Edit required");string folder="Assets/Art/Environment/JourneyV48/DamR99/",d="Test/KetQua/journey_v48/dam_r99/";File.Copy("Test/Output/journey_v48/DamR99/OngDaoDamR99.fbx",folder+"OngDaoDamR99.fbx",true);AssetDatabase.ImportAsset(folder+"OngDaoDamR99.fbx",ImportAssetOptions.ForceUpdate);var imported=AssetDatabase.LoadAssetAtPath<GameObject>(folder+"OngDaoDamR99.fbx").GetComponentsInChildren<MeshFilter>(true).ToDictionary(x=>x.name);var log=new System.Text.StringBuilder();foreach(var l in File.ReadAllLines(d+"expected-triangles.tsv")){var e=l.Split('\t');int actual=imported[e[0]].sharedMesh.triangles.Length/3;log.AppendLine(e[0]+" expected="+e[1]+" imported="+actual);if(actual!=int.Parse(e[1]))throw new System.Exception("Import triangle mismatch "+e[0]+" "+actual+" vs "+e[1]);}
foreach(var l in File.ReadAllLines(d+"integration.tsv")){var e=l.Split('\t');if(!e[1].StartsWith("R97_STAIR_")&&!e[1].StartsWith("D99_FLOWERS_")&&!e[1].StartsWith("R99_SUPPORT_"))continue;var g=sc.GetRootGameObjects().SelectMany(x=>x.GetComponentsInChildren<Transform>(true)).First(t=>P(t)==e[2]);var src=imported[e[0]];var mesh=Object.Instantiate(src.sharedMesh);var v=mesh.vertices;var n=mesh.normals;for(int i=0;i<v.Length;i++){v[i]=g.InverseTransformPoint(src.transform.TransformPoint(v[i]));n[i]=g.localToWorldMatrix.transpose.MultiplyVector(src.transform.worldToLocalMatrix.transpose.MultiplyVector(n[i])).normalized;}mesh.vertices=v;mesh.normals=n;mesh.RecalculateBounds();mesh.RecalculateTangents();var dst=AssetDatabase.LoadAssetAtPath<UnityEngine.Mesh>(folder+e[1]+"_BoundR99.asset");if(dst){mesh.name=dst.name;EditorUtility.CopySerialized(mesh,dst);Object.DestroyImmediate(mesh);}else{mesh.name=e[1]+"_BoundR99";AssetDatabase.CreateAsset(mesh,folder+mesh.name+".asset");dst=mesh;}if(g.GetComponent<MeshFilter>()&&!e[1].StartsWith("R99_SUPPORT_"))g.GetComponent<MeshFilter>().sharedMesh=dst;dst.UploadMeshData(false);var c=g.GetComponent<MeshCollider>();if(c){c.sharedMesh=null;c.sharedMesh=dst;}EditorUtility.SetDirty(dst);if(c)EditorUtility.SetDirty(c);}
AssetDatabase.SaveAssets();EditorSceneManager.MarkSceneDirty(sc);EditorSceneManager.SaveScene(sc);File.WriteAllText(d+"unity-import-triangles.txt",log.ToString());result.Log("Imported triangle counts agree with source; R97 stair and supplemental flower bindings refreshed without moving anchors.");}}
