{
 public void Execute(ExecutionResult result)
 {
  if(!EditorApplication.isPlaying)throw new Exception("Play required");var s=UnityEngine.Object.FindFirstObjectByType<InvestigationSession>();int initialLeg=0;if(s.InspectionCount==1&&s.Has("khai")&&Vector2.Distance(new Vector2(s.movement.transform.position.x,s.movement.transform.position.z),new Vector2(773.6127f,289.8756f))<.3f)initialLeg=1;else if(Vector2.Distance(new Vector2(s.movement.transform.position.x,s.movement.transform.position.z),new Vector2(774,286))>.3f||s.InspectionCount!=0)throw new Exception("Normal start or exact unplaced continuation required");
  s.SaveDirectoryOverride=Path.GetFullPath("Test/KetQua/investigation_r102/route-profile");s.ClosePage();
  var settings=InputSystem.settings;if(AssetDatabase.Contains(settings))throw new Exception("Runtime InputSettings required");string prior=JsonUtility.ToJson(settings);settings.editorInputBehaviorInPlayMode=InputSettings.EditorInputBehaviorInPlayMode.AllDeviceInputAlwaysGoesToGameView;settings.backgroundBehavior=InputSettings.BackgroundBehavior.IgnoreFocus;
  var kb=InputSystem.AddDevice<Keyboard>("Investigation102_SyntheticKeyboard");var mouse=InputSystem.AddDevice<Mouse>("Investigation102_SyntheticMouse");var map=InputSystem.ListEnabledActions().First(a=>a.name=="Move"&&a.actionMap.name=="Gameplay").actionMap;var devices=map.devices;map.devices=new InputDevice[]{kb,mouse};
  var goals=new[]{
   (new Vector2(774,286),"khai",new Vector3(776.7f,6.1f,286)),
   (new Vector2(773.6f,290),"henri",new Vector3(772.05f,5.956f,290.14f)),
   (new Vector2(774,289.7f),"r03",new Vector3(772.6f,5.7f,291)),

   (new Vector2(770.1f,292.7f),"",Vector3.zero),
   (new Vector2(773.8f,293),"slide",new Vector3(774.8f,8.075f,291.75f)),

   (new Vector2(774.8f,293.2f),"phone",new Vector3(776.3f,8.895f,293.1f)),
   (new Vector2(774.8f,293.2f),"r12",new Vector3(776.2f,9,293.8f)),
   (new Vector2(774.6f,293),"mechanism",new Vector3(774.75f,9.05f,291.5f)),
   (new Vector2(770.1f,292.7f),"",Vector3.zero),
   (new Vector2(770.1f,285.7f),"",Vector3.zero),
   (new Vector2(774,286),"hook",new Vector3(775.26f,6.32f,285.51f)),
   (new Vector2(774,286),"fabric",new Vector3(775.55f,6.345f,285.55f)),
   (new Vector2(774,282),"",Vector3.zero),
   (new Vector2(780,276),"",Vector3.zero),

   (new Vector2(780,277),"",Vector3.zero),

   (new Vector2(774,286),"",Vector3.zero),

   if(now>hold&&leg<goals.Length&&!finished){var g=goals[leg];var p=s.movement.transform.position;

     float yaw=Mathf.Atan2(aim.x,aim.z)*Mathf.Rad2Deg;float turn=Mathf.DeltaAngle(s.movement.transform.eulerAngles.y,yaw);float pitch=arrived&&g.Item2!=""?Mathf.Atan2(-aim.y,new Vector2(aim.x,aim.z).magnitude)*Mathf.Rad2Deg:8;
     float dp=Mathf.DeltaAngle(s.eye.transform.localEulerAngles.x,pitch);delta=new Vector2(Mathf.Clamp(turn,-4,4),-Mathf.Clamp(dp,-4,4))/(s.look.mouseSensitivity*.0002f);
     if(!arrived){if(Mathf.Abs(turn)<6)keys=new KeyboardState(Key.W);}
     else if(g.Item2==""){log.AppendLine("WAYPOINT "+leg+" p="+p.ToString("F4")+" grounded="+s.movement.controller.isGrounded);leg++;hold=now+.3;legStart=now;}

      else if(now-legStart>8){log.AppendLine("BLOCKED_RAY "+g.Item2+" hit="+(hit?ray.collider.name:"none")+" id="+(item?item.evidenceId:"none")+" p="+p.ToString("F4"));finished=true;}
     }
    }
   }
   InputSystem.QueueStateEvent(kb,keys);InputSystem.QueueStateEvent(mouse,new MouseState{delta=delta});
  };

  EditorApplication.CallbackFunction update=null;update=()=>{
   double now=EditorApplication.timeSinceStartup,t=now-begin;if(!EditorApplication.isPlaying){EditorApplication.update-=update;InputSystem.onBeforeUpdate-=input;return;}

   if(leg<goals.Length&&!finished&&t<210&&now-legStart<35)return;
   InputSystem.onBeforeUpdate-=input;InputSystem.ResetDevice(kb);map.devices=devices;InputSystem.RemoveDevice(kb);InputSystem.RemoveDevice(mouse);JsonUtility.FromJsonOverwrite(prior,settings);EditorApplication.update-=update;s.Save();log.AppendLine("END "+leg+"/"+goals.Length+" seconds="+t+" data="+JsonUtility.ToJson(s.data));File.WriteAllText("Test/KetQua/investigation_r102/route-motion.txt",log.ToString());
  };EditorApplication.update+=update;result.Log("Started continuous controller route / E ray interactions. 20 stops, max210s; no placement. Final report generated after actual traversal.");
 }
}
