using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using System;using System.Linq;using ConRoi.Investigation.V36.R104;
internal class CommandScript:IRunCommand{public void Execute(ExecutionResult r){if(EditorApplication.isPlaying)throw new Exception("Edit required");var sc=EditorSceneManager.GetActiveScene();if(!sc.path.EndsWith("r104.unity"))throw new Exception("r104 only");const string p="Assets/Art/InvestigationV36/R104/CuffComparison.fbx";AssetDatabase.ImportAsset(p,ImportAssetOptions.ForceSynchronousImport);var source=AssetDatabase.LoadAssetAtPath<GameObject>(p);var mesh=source.GetComponentInChildren<MeshFilter>().sharedMesh;var item=Object.FindObjectsByType<InvestigationInteractable>(FindObjectsInactive.Include,FindObjectsSortMode.None).Single(x=>x.evidenceId=="coat");var f=item.GetComponentInChildren<MeshFilter>();if(!f)throw new Exception("coat filter absent");f.sharedMesh=mesh;var old=f.GetComponent<Renderer>();var mat=AssetDatabase.LoadAssetAtPath<Material>("Assets/Art/InvestigationV36/R104/Materials/Khai_cuff_denim.mat");if(!mat){mat=new Material(Shader.Find("Universal Render Pipeline/Lit"));mat.color=new Color(.03f,.14f,.25f);AssetDatabase.CreateAsset(mat,"Assets/Art/InvestigationV36/R104/Materials/Khai_cuff_denim.mat");}old.sharedMaterials=Enumerable.Repeat(mat,mesh.subMeshCount).ToArray();foreach(var c in f.GetComponents<MeshCollider>())c.sharedMesh=mesh;var box=f.GetComponent<BoxCollider>();if(box){box.center=mesh.bounds.center;box.size=Vector3.Max(mesh.bounds.size,new Vector3(.005f,.005f,.005f));}var s=Object.FindFirstObjectByType<InvestigationSession>();s.data=new InvestigationSave();EditorSceneManager.MarkSceneDirty(sc);EditorSceneManager.SaveScene(sc);AssetDatabase.SaveAssets();r.Log("Cuff sample integrated into existing coat root "+item.name+" mesh="+mesh.name);}}
