using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using System.IO;using System.Linq;
internal class CommandScript:IRunCommand{[System.Serializable]public class Row{public string alias,key,path;}[System.Serializable]public class Rows{public Row[] rows;}static string P(Transform t)=>t.parent?P(t.parent)+"/"+t.name:t.name;
public void Execute(ExecutionResult result){if(EditorApplication.isPlaying||EditorApplication.isCompiling)throw new System.Exception("Edit only");var sc=EditorSceneManager.GetActiveScene();if(!sc.path.EndsWith("r99.unity")||sc.isDirty)throw new System.Exception("Clean r99 required");string dst="Assets/Scenes/JourneyPresentation_v48_r100.unity",folder="Assets/Art/Environment/JourneyV48/DamR100/",d="Test/KetQua/journey_v48/map_r99_closeout/";if(File.Exists(dst)||Directory.Exists(folder))throw new System.Exception("R100 already exists");var rows=File.ReadAllLines(d+"r100-integration.tsv").Select(l=>l.Split('\t')).Select(e=>new Row{alias=e[0],key=e[1],path=e[2]}).ToArray();var ts=sc.GetRootGameObjects().SelectMany(g=>g.GetComponentsInChildren<Transform>(true)).ToArray();foreach(var row in rows){var t=ts.Single(x=>P(x)==row.path);if(!t.GetComponent<MeshFilter>()||!t.GetComponent<MeshRenderer>())throw new System.Exception("Missing renderer "+row.path);if(t.GetComponent<MeshCollider>())throw new System.Exception("Unexpected collider, stop and reconcile "+row.path);}
Directory.CreateDirectory(folder);File.Copy("Test/Output/journey_v48/DamR100/OngDaoDamR100_Delta.fbx",folder+"OngDaoDamR100_Delta.fbx",false);AssetDatabase.Refresh();var imp=(ModelImporter)AssetImporter.GetAtPath(folder+"OngDaoDamR100_Delta.fbx");imp.isReadable=true;imp.importAnimation=false;imp.importNormals=ModelImporterNormals.Import;imp.importTangents=ModelImporterTangents.CalculateMikk;imp.meshCompression=ModelImporterMeshCompression.Off;imp.optimizeMeshVertices=false;imp.optimizeMeshPolygons=false;imp.weldVertices=false;imp.materialImportMode=ModelImporterMaterialImportMode.None;imp.SaveAndReimport();var imported=AssetDatabase.LoadAssetAtPath<GameObject>(folder+"OngDaoDamR100_Delta.fbx").GetComponentsInChildren<MeshFilter>(true).ToDictionary(x=>x.name);EditorSceneManager.SaveScene(sc,dst);var log=new System.Text.StringBuilder();
foreach(var row in rows){var t=ts.Single(x=>P(x)==row.path);var f=t.GetComponent<MeshFilter>();var r=t.GetComponent<MeshRenderer>();var src=imported[row.alias];var mesh=Object.Instantiate(src.sharedMesh);mesh.name=row.key+"_BoundR100";var v=mesh.vertices;var n=mesh.normals;for(int i=0;i<v.Length;i++){v[i]=t.InverseTransformPoint(src.transform.TransformPoint(v[i]));if(n.Length==v.Length)n[i]=t.localToWorldMatrix.transpose.MultiplyVector(src.transform.worldToLocalMatrix.transpose.MultiplyVector(n[i])).normalized;}mesh.vertices=v;mesh.normals=n;mesh.RecalculateBounds();if(mesh.uv.Length==v.Length)mesh.RecalculateTangents();mesh.UploadMeshData(false);AssetDatabase.CreateAsset(mesh,folder+mesh.name+".asset");f.sharedMesh=mesh;EditorUtility.SetDirty(f);PrefabUtility.RecordPrefabInstancePropertyModifications(f);if(r.sharedMaterials.Any(x=>!x))throw new System.Exception("Material missing");log.AppendLine(row.path+" -> "+AssetDatabase.GetAssetPath(mesh)+" verts="+mesh.vertexCount+" renderer="+r.enabled+" collider=none layer="+t.gameObject.layer);}
AssetDatabase.SaveAssets();EditorSceneManager.MarkSceneDirty(sc);EditorSceneManager.SaveScene(sc);File.WriteAllText(d+"r100-integration.txt",log.ToString());result.Log("R100 derivative saved: eight fresh delta meshes; all walking surfaces/colliders, Bảo/gameplay/migration untouched; Build Settings unchanged.");}}
