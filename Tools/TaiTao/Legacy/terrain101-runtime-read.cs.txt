using UnityEngine;using UnityEditor;using System.IO;using System.Linq;
internal class CommandScript:IRunCommand{static string P(Transform t)=>t.parent?P(t.parent)+"/"+t.name:t.name;public void Execute(ExecutionResult result){var ts=Object.FindObjectsByType<Transform>(FindObjectsInactive.Include,FindObjectsSortMode.None);var b=new System.Text.StringBuilder();foreach(var line in File.ReadAllLines("Test/KetQua/journey_v48/terrain_r101/final-integration.tsv")){var e=line.Split('\t');var t=ts.Single(t=>P(t)==e[2]);var f=t.GetComponent<MeshFilter>();var c=t.GetComponent<MeshCollider>();var expected=AssetDatabase.LoadAssetAtPath<UnityEngine.Mesh>("Assets/Art/Environment/JourneyV48/TerrainR101/"+e[1]+"_IntegratedR101.asset");b.AppendLine(e[2]+" expected="+AssetDatabase.GetAssetPath(expected)+" filter="+AssetDatabase.GetAssetPath(f.sharedMesh)+" collider="+AssetDatabase.GetAssetPath(c.sharedMesh)+" same="+(f.sharedMesh==c.sharedMesh)+" hidden="+!t.GetComponent<MeshRenderer>().enabled);foreach(var pair in new[]{f.sharedMesh,c.sharedMesh}){float max=999;bool tri=false;if(pair&&pair.isReadable&&expected&&pair.vertexCount==expected.vertexCount){var a=pair.vertices;var v=expected.vertices;max=0;for(int i=0;i<a.Length;i++)max=Mathf.Max(max,Vector3.Distance(a[i],v[i]));tri=pair.triangles.SequenceEqual(expected.triangles);}b.AppendLine("name="+(pair?pair.name:"null")+" verts="+(pair?pair.vertexCount:0)+" readable="+(pair&&pair.isReadable)+" maxLocalVertexDelta="+max+" trianglesSame="+tri);}}File.WriteAllText("Test/KetQua/journey_v48/terrain_r101/runtime-geometry-read.txt",b.ToString());result.Log(b.ToString());}}
