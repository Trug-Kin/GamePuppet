using UnityEngine;using UnityEditor;using UnityEngine.InputSystem;using UnityEngine.InputSystem.LowLevel;using System;using System.IO;using System.Text;using ConRoi.Investigation.V36.R103;
internal class CommandScript:IRunCommand{public void Execute(ExecutionResult r){var s=Object.FindFirstObjectByType<InvestigationSession>();if(!EditorApplication.isPlaying)throw new Exception("Play required");s.SaveDirectoryOverride=Path.GetFullPath("Test/KetQua/investigation_r103/input-profile");var w=EditorWindow.GetWindow(Type.GetType("UnityEditor.GameView,UnityEditor"));s.OpenPage("cipher");var settings=InputSystem.settings;if(AssetDatabase.Contains(settings))throw new Exception("Runtime input settings required");string prior=JsonUtility.ToJson(settings);settings.editorInputBehaviorInPlayMode=InputSettings.EditorInputBehaviorInPlayMode.AllDeviceInputAlwaysGoesToGameView;settings.backgroundBehavior=InputSettings.BackgroundBehavior.IgnoreFocus;var kb=InputSystem.AddDevice<Keyboard>("R103_EdgeKeys");var mouse=InputSystem.AddDevice<Mouse>("R103_InspectMouse");var b=new StringBuilder("Synthetic InputSystem key/mouse events. API opens known pages and selects test view only. Not controller route.\n");int i=0;double next=EditorApplication.timeSinceStartup+.7;Camera preview=null;Vector3 before=Vector3.zero;EditorApplication.CallbackFunction f=null;Action<bool,string> check=(ok,t)=>{b.AppendLine((ok?"PASS ":"FAIL ")+t);if(!ok)throw new Exception(t);};f=()=>{if(EditorApplication.timeSinceStartup<next)return;next=EditorApplication.timeSinceStartup+.5;try{switch(i++){
case 0:var p=new Vector2(286*Screen.width/1200f/1.25f,484*Screen.height/800f/1.25f+55);w.SendEvent(new Event{type=EventType.MouseDown,button=0,mousePosition=p});w.SendEvent(new Event{type=EventType.MouseUp,button=0,mousePosition=p});break;
case 1:InputSystem.QueueStateEvent(kb,new KeyboardState(Key.R));break;
case 2:InputSystem.QueueStateEvent(kb,new KeyboardState());check(s.Page=="cipher","raw R while editing stays in cipher");break;
case 3:InputSystem.QueueStateEvent(kb,new KeyboardState(Key.P));break;
case 4:InputSystem.QueueStateEvent(kb,new KeyboardState());check(s.Page=="cipher","raw P while editing stays in cipher");break;
case 5:InputSystem.QueueStateEvent(kb,new KeyboardState(Key.Escape,Key.E));break;
case 6:InputSystem.QueueStateEvent(kb,new KeyboardState());check(s.Page==""&&!s.Paused,"Esc closes page; same-frame E does not open world action");break;
case 7:InputSystem.QueueStateEvent(kb,new KeyboardState(Key.Escape));break;
case 8:InputSystem.QueueStateEvent(kb,new KeyboardState());check(s.Paused&&Time.timeScale==0,"Esc pauses gameplay");break;
case 9:InputSystem.QueueStateEvent(kb,new KeyboardState(Key.Escape));break;
case 10:InputSystem.QueueStateEvent(kb,new KeyboardState());check(!s.Paused&&Time.timeScale==1,"Esc resumes");s.OpenEvidence("shoeprint");break;
case 11:preview=GameObject.Find("Evidence preview camera").GetComponent<Camera>();before=preview.transform.position;InputSystem.QueueStateEvent(mouse,new MouseState{buttons=1,delta=new Vector2(120,-50)});break;
case 12:InputSystem.QueueStateEvent(mouse,new MouseState());check(Vector3.Distance(before,preview.transform.position)>.01f,"inspection mouse rotates actual preview camera");before=preview.transform.position;break;
case 13:InputSystem.QueueStateEvent(mouse,new MouseState{scroll=new Vector2(0,120)});break;
case 14:InputSystem.QueueStateEvent(mouse,new MouseState());check(Vector3.Distance(before,preview.transform.position)>.01f,"inspection wheel zoom changes preview camera");ScreenCapture.CaptureScreenshot("Test/BangChung/investigation_r103/inspection-input.png");break;
default:s.ClosePage();InputSystem.RemoveDevice(kb);InputSystem.RemoveDevice(mouse);JsonUtility.FromJsonOverwrite(prior,settings);EditorApplication.update-=f;File.WriteAllText("Test/KetQua/investigation_r103/input-edges.txt",b.ToString());break;
}}catch(Exception e){InputSystem.RemoveDevice(kb);InputSystem.RemoveDevice(mouse);JsonUtility.FromJsonOverwrite(prior,settings);EditorApplication.update-=f;File.WriteAllText("Test/KetQua/investigation_r103/input-edges.txt",b+"FAIL "+e);}};EditorApplication.update+=f;r.Log("Bounded raw key/inspection checks");}}
