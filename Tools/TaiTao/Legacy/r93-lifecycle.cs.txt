System.Action input=()=>{if(InputState.currentUpdateType!=InputUpdateType.Dynamic)return;double t=EditorApplication.timeSinceStartup-begin;InputSystem.QueueStateEvent(kb,(t>.4&&t<.6)||(t>2.5&&t<2.7)?new KeyboardState(Key.Escape):t>1&&t<2?new KeyboardState(Key.W):new KeyboardState());};InputSystem.onBeforeUpdate+=input;
System.Action<string> count=(tag)=>{var ts=Object.FindObjectsByType<Transform>(FindObjectsInactive.Include,FindObjectsSortMode.None);var root=ts.Single(x=>x.name=="Langfarm_R93");b.AppendLine(tag+" region="+s.data.region+" new="+ts.Count(x=>x.name=="Langfarm_R93")+" old="+ts.Count(x=>new[]{"R54_Langfarm_Facade_ESTIMATED","LangfarmReturn_R86","LangfarmFasciaRemain_R86"}.Contains(x.name))+" meshes="+root.GetComponentsInChildren<MeshFilter>(true).Length+" MC="+root.GetComponentsInChildren<MeshCollider>(true).Length+" FBX="+string.Join(";",root.GetComponentsInChildren<MeshFilter>(true).Select(x=>AssetDatabase.GetAssetPath(x.sharedMesh)).Distinct()));};count("start");
