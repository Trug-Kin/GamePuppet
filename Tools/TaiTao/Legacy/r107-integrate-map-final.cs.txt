internal class CommandScript:IRunCommand{
string P(Transform t)=>t.parent?P(t.parent)+"/"+t.name:t.name;
Vector3Int Key(Vector3 p)=>new Vector3Int(Mathf.FloorToInt(p.x*50),Mathf.FloorToInt(p.y*50),Mathf.FloorToInt(p.z*50));
public void Execute(ExecutionResult r){var sc=EditorSceneManager.GetActiveScene();if(EditorApplication.isPlaying||sc.isDirty||!sc.path.EndsWith("r107.unity"))throw new System.Exception("Clean R107 required");string folder="Assets/Art/Environment/JourneyV48/TerrainR107";Directory.CreateDirectory(folder);var fbx=folder+"/TerrainR107_Final.fbx";if(!File.Exists(fbx))File.Copy("Test/KetQua/investigation_r107/export/TerrainR107_Final.fbx",fbx);AssetDatabase.Refresh();var imp=(ModelImporter)AssetImporter.GetAtPath(fbx);imp.isReadable=true;imp.importAnimation=false;imp.meshCompression=ModelImporterMeshCompression.Off;imp.optimizeMeshVertices=false;imp.optimizeMeshPolygons=false;imp.weldVertices=false;imp.materialImportMode=ModelImporterMaterialImportMode.None;imp.SaveAndReimport();var srcs=AssetDatabase.LoadAssetAtPath<GameObject>(fbx).GetComponentsInChildren<MeshFilter>(true).ToDictionary(f=>f.name);var ts=sc.GetRootGameObjects().SelectMany(g=>g.GetComponentsInChildren<Transform>(true)).ToArray();var paths=new[]{"R20_Whole_Source_Export/Surface_176","R20_Whole_Source_Export/Surface_104","LakeGateMarketV32/Context_124","R20_Whole_Source_Export/Surface_142"};var b=new StringBuilder("Source Blender -> FBX -> orientation transfer to original bound topology. Original coordinates, UVs, non-target triangles retained.\n");
for(int k=0;k<paths.Length;k++){var t=ts.Single(t=>P(t)==paths[k]);var mf=t.GetComponent<MeshFilter>();var mc=t.GetComponent<MeshCollider>();var old=k==2?mc.sharedMesh:mf.sharedMesh;var src=srcs["T107_"+(k==3?4:k)];var sv=src.sharedMesh.vertices;var st=src.sharedMesh.triangles;var grid=new Dictionary<Vector3Int,List<(Vector3 p,Vector3 n)>>();for(int i=0;i<st.Length;i+=3){var a=src.transform.TransformPoint(sv[st[i]]);var c=src.transform.TransformPoint(sv[st[i+1]]);var d=src.transform.TransformPoint(sv[st[i+2]]);var p=(a+c+d)/3;var key=Key(p);if(!grid.ContainsKey(key))grid[key]=new List<(Vector3,Vector3)>();grid[key].Add((p,(Vector3.Cross(c-a,d-a)*1000000f).normalized));}
var m=Object.Instantiate(old);m.name=old.name+"_FinalWindingR107";var vv=m.vertices;var tr=m.triangles;var oldNormals=m.normals;var changedVerts=new HashSet<int>();int flipped=0,unmatched=0;for(int i=0;i<tr.Length;i+=3){var a=t.TransformPoint(vv[tr[i]]);var c=t.TransformPoint(vv[tr[i+1]]);var d=t.TransformPoint(vv[tr[i+2]]);var p=(a+c+d)/3;var n=(Vector3.Cross(c-a,d-a)*1000000f).normalized;if(p.x<560||p.x>720||p.z<320||p.z>460||p.y<=-.05f||n.y>=-.2f)continue;float best=.0004f;Vector3 normal=Vector3.zero;var q=Key(p);for(int x=-1;x<=1;x++)for(int y=-1;y<=1;y++)for(int z=-1;z<=1;z++)if(grid.TryGetValue(q+new Vector3Int(x,y,z),out var list))foreach(var item in list){float ds=(item.p-p).sqrMagnitude;if(ds<best){best=ds;normal=item.n;}}if(normal==Vector3.zero){float nearest=100;Vector3 near=Vector3.zero;foreach(var bucket in grid.Values)foreach(var item in bucket){float ds=(item.p-p).sqrMagnitude;if(ds<nearest){nearest=ds;near=item.p;}}throw new System.Exception("Unmatched triangle="+i/3+" at="+p.ToString("F5")+" nearest="+near.ToString("F5")+" delta="+Mathf.Sqrt(nearest));}if(normal.y>.2f){int swap=tr[i+1];tr[i+1]=tr[i+2];tr[i+2]=swap;changedVerts.Add(tr[i]);changedVerts.Add(tr[i+1]);changedVerts.Add(tr[i+2]);flipped++;}}
if(unmatched>0)throw new System.Exception("Unmatched source triangles "+paths[k]+" "+unmatched);m.triangles=tr;m.RecalculateNormals();var normals=m.normals;if(oldNormals.Length==normals.Length)for(int i=0;i<normals.Length;i++)if(!changedVerts.Contains(i))normals[i]=oldNormals[i];m.normals=normals;m.RecalculateBounds();if(m.uv.Length==vv.Length)m.RecalculateTangents();if(!vv.SequenceEqual(old.vertices))throw new System.Exception("Unexpected vertex move");AssetDatabase.CreateAsset(m,folder+"/"+m.name+".asset");if(k==2){mc.sharedMesh=m;EditorUtility.SetDirty(mc);}else{mf.sharedMesh=m;EditorUtility.SetDirty(mf);if(mc){mc.sharedMesh=m;EditorUtility.SetDirty(mc);}}b.AppendLine(paths[k]+" old="+AssetDatabase.GetAssetPath(old)+" new="+AssetDatabase.GetAssetPath(m)+" flipped="+flipped+" unmatched="+unmatched+" vertexDelta=0 UVunchanged="+m.uv.SequenceEqual(old.uv));}
AssetDatabase.SaveAssets();EditorSceneManager.MarkSceneDirty(sc);EditorSceneManager.SaveScene(sc);File.WriteAllText("Test/KetQua/investigation_r107/map-integration-final.txt",b.ToString());r.Log(b.ToString());}}
