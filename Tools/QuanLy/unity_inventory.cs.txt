using UnityEngine;using UnityEditor;using UnityEngine.SceneManagement;using System;using System.IO;using System.Linq;using System.Collections.Generic;using ConRoi.Investigation.V36.R107;
internal class CommandScript:IRunCommand{
[Serializable]class Entry{public string path,guid;public bool r107,build,resource,special;}
[Serializable]class Report{public string unity,scene;public bool play,dirty,compiling;public string[] dirtyScenes,compilationSources;public Entry[] assets;public string liveProfile;}
public void Execute(ExecutionResult r){var sc=SceneManager.GetActiveScene();var current="Assets/Scenes/JourneyPresentation_v48_r107.unity";var deps=new HashSet<string>(AssetDatabase.GetDependencies(current,true));var builds=new HashSet<string>(EditorBuildSettings.scenes.SelectMany(s=>AssetDatabase.GetDependencies(s.path,true)));var all=AssetDatabase.GetAllAssetPaths().Where(p=>p.StartsWith("Assets/")).ToArray();var dirty=Enumerable.Range(0,SceneManager.sceneCount).Select(SceneManager.GetSceneAt).Where(s=>s.isDirty).Select(s=>s.path).ToArray();var session=UnityEngine.Object.FindFirstObjectByType<InvestigationSession>();var report=new Report{unity=Application.unityVersion,scene=sc.path,play=EditorApplication.isPlaying,dirty=sc.isDirty,compiling=EditorApplication.isCompiling,dirtyScenes=dirty,compilationSources=UnityEditor.Compilation.CompilationPipeline.GetAssemblies().SelectMany(a=>a.sourceFiles).Distinct().ToArray(),liveProfile=session?session.SavePath:"none",assets=all.Select(p=>new Entry{path=p,guid=AssetDatabase.AssetPathToGUID(p),r107=deps.Contains(p),build=builds.Contains(p),resource=p.Contains("/Resources/"),special=p.Contains("/StreamingAssets/")||p.Contains("Addressable")}).ToArray()};Directory.CreateDirectory("Docs/QuanLy");File.WriteAllText("Docs/QuanLy/unity-dependencies.json",JsonUtility.ToJson(report,true));File.WriteAllLines("Docs/QuanLy/unity-assets.tsv",new[]{"path\tguid\tr107\tbuild\tresource\tspecial"}.Concat(report.assets.Select(a=>a.path+"\t"+a.guid+"\t"+a.r107+"\t"+a.build+"\t"+a.resource+"\t"+a.special)));if(EditorApplication.isPlaying&&session)File.WriteAllText("Docs/QuanLy/live-state-preserved.json",JsonUtility.ToJson(session.data,true));r.Log("Read-only inventory: scene="+sc.path+" play="+report.play+" dirty="+report.dirty+" assets="+all.Length+" currentDeps="+deps.Count+" compileSources="+report.compilationSources.Length+". No SaveScene/Stop/input.");}}
