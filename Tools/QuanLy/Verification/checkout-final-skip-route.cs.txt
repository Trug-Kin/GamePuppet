using UnityEngine;using UnityEditor;using UnityEngine.InputSystem;using UnityEngine.InputSystem.LowLevel;using System;using System.IO;using System.Linq;using System.Text;using ConRoi.Investigation.V36.R107;
internal class CommandScript:IRunCommand{public void Execute(ExecutionResult result){var s=UnityEngine.Object.FindFirstObjectByType<InvestigationSession>();if(s.Page!=""||s.InspectionCount!=0)throw new Exception("Fresh intro required; no reset allowed");var w=EditorWindow.GetWindow(Type.GetType("UnityEditor.GameView,UnityEditor"));w.Focus();var settings=InputSystem.settings;if(AssetDatabase.Contains(settings))throw new Exception("Runtime settings required");string prior=JsonUtility.ToJson(settings);settings.editorInputBehaviorInPlayMode=InputSettings.EditorInputBehaviorInPlayMode.AllDeviceInputAlwaysGoesToGameView;settings.backgroundBehavior=InputSettings.BackgroundBehavior.IgnoreFocus;var kb=InputSystem.AddDevice<Keyboard>("R107_route_keyboard");var mouse=InputSystem.AddDevice<Mouse>("R107_route_mouse");kb.MakeCurrent();mouse.MakeCurrent();InputActionMap map=null;UnityEngine.InputSystem.Utilities.ReadOnlyArray<InputDevice>? devices=null;
var goals=new[]{new Vector2(774,286),new Vector2(770.1f,285.7f),new Vector2(770.1f,292.7f),new Vector2(773.8f,293)};
double begin=EditorApplication.timeSinceStartup,hold=begin+7,last=0,legStart=begin;int leg=0;bool clicked=true,done=false;var log=new StringBuilder("Synthetic GameView mouse + InputSystem controller. Normal saved spawn; no teleport, API page open, evidence injection, or API Observe.\n");
Action input=()=>{if(InputState.currentUpdateType!=InputUpdateType.Dynamic)return;double now=EditorApplication.timeSinceStartup;var keys=new KeyboardState();Vector2 delta=Vector2.zero;
if(!s.Modal&&map==null){map=InputSystem.ListEnabledActions().First(a=>a.name=="Move"&&a.actionMap.name=="Gameplay").actionMap;devices=map.devices;map.devices=new InputDevice[]{kb,mouse};}
if(now>hold&&s.Page==""&&leg<goals.Length){var p=s.movement.transform.position;var d=goals[leg]-new Vector2(p.x,p.z);bool arrived=d.magnitude<.16f;var aim=arrived&&leg==goals.Length-1?new Vector3(774.835f,8.075f,292.02f)-s.eye.transform.position:new Vector3(d.x,0,d.y);float yaw=Mathf.Atan2(aim.x,aim.z)*Mathf.Rad2Deg;float turn=Mathf.DeltaAngle(s.movement.transform.eulerAngles.y,yaw);float pitch=arrived&&leg==goals.Length-1?Mathf.Atan2(-aim.y,new Vector2(aim.x,aim.z).magnitude)*Mathf.Rad2Deg:8;float dp=Mathf.DeltaAngle(s.eye.transform.localEulerAngles.x,pitch);delta=new Vector2(Mathf.Clamp(turn,-3,3),-Mathf.Clamp(dp,-3,3))/(s.look.mouseSensitivity*.0002f);
if(!arrived){if(Mathf.Abs(turn)<6)keys=new KeyboardState(Key.W);}else if(leg<goals.Length-1){log.AppendLine("WAYPOINT="+leg+" p="+p+" grounded="+s.movement.controller.isGrounded);leg++;hold=now+2;legStart=now;}else if(Mathf.Abs(turn)<2&&Mathf.Abs(dp)<2){bool hit=s.TryInteractionHit(out var ray);var item=hit?ray.collider.GetComponentInParent<InvestigationInteractable>():null;if(item&&(item.evidenceId=="slide"||item.evidenceId=="shoeprint")){keys=new KeyboardState(Key.E);done=true;log.AppendLine("E_RAY="+item.evidenceId+" distance="+ray.distance+" p="+p);}else if(now-legStart>15){log.AppendLine("FAIL ray="+(hit?ray.collider.name:"none"));done=true;}}}
InputSystem.QueueStateEvent(kb,keys);InputSystem.QueueStateEvent(mouse,new MouseState{delta=delta});};InputSystem.onBeforeUpdate+=input;
EditorApplication.CallbackFunction update=null;update=()=>{double now=EditorApplication.timeSinceStartup,t=now-begin;if(!clicked&&t>5){var pos=new Vector2(600*Screen.width/1200f/1.25f,390*Screen.height/800f/1.25f+55);w.SendEvent(new Event{type=EventType.MouseDown,button=0,mousePosition=pos});w.SendEvent(new Event{type=EventType.MouseUp,button=0,mousePosition=pos});clicked=true;log.AppendLine("UI start button, introAccepted="+s.data.introAccepted);}
if(t-last>.3){last=t;File.WriteAllText("Test/Output/CleanCheckout/skip-route-progress.txt",log+"t="+t+" leg="+leg+" p="+s.movement.transform.position+" page="+s.Page);}
if(!(done&&s.Page=="inspect")&&t<100)return;
InputSystem.onBeforeUpdate-=input;InputSystem.ResetDevice(kb);if(map!=null)map.devices=devices;InputSystem.RemoveDevice(kb);InputSystem.RemoveDevice(mouse);JsonUtility.FromJsonOverwrite(prior,settings);EditorApplication.update-=update;ScreenCapture.CaptureScreenshot("Test/Output/CleanCheckout/skip-first-inspection-1080.png");log.AppendLine((s.Page=="inspect"?"PASS":"FAIL")+" floor reached and opened through E/raycast; records="+s.InspectionCount+" objective="+s.CurrentObjective);File.WriteAllText("Test/Output/CleanCheckout/skip-route.txt",log.ToString());};EditorApplication.update+=update;result.Log("Started onboarding route from intro; keyboard/mouse controller only");}}
