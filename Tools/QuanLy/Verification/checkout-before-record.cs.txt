internal class CommandScript:IRunCommand{public void Execute(ExecutionResult r){

var w=EditorWindow.GetWindow(Type.GetType("UnityEditor.GameView,UnityEditor"));w.Focus();var b=new StringBuilder("Synthetic GameView click + InputSystem keys; state assertions via read-only API.\n");b.AppendLine("resolution="+Screen.width+"x"+Screen.height+" profile="+s.SavePath);var p=s.movement.transform.position;
